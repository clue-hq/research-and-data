- Produces:
  ```python
  @dataclass(frozen=True)
  class KeyClaim: field_key: str; alternatives: tuple[str, ...]; value: JsonValue; source_row_id: str | None
  @dataclass(frozen=True)
  class Match: key: KeyClaim; verdict: Literal["exact", "partial", "field_only", "missing"]; run_claim_id: str | None; run_value: JsonValue | None
  @dataclass(frozen=True)
  class RecallReport:
      account: str; keys: int; exact: int; partial: int; field_only: int; missing: int
      matches: tuple[Match, ...]
      def recall(self) -> float            # (exact + partial) / keys
      def by_section(self) -> dict[str, tuple[int, int]]   # field prefix ("core", "account.resonance", …) → (hits, keys)
      def markdown(self) -> str
  def normalise_value(value: JsonValue) -> str      # lower-cased, whitespace-collapsed, digits kept, urls stripped of scheme/www
  def values_agree(key_value: JsonValue, run_value: JsonValue) -> Literal["exact", "partial", "no"]
      # exact: normalised strings equal, or numbers equal, or the key value is contained in the run value's normalised text (a longer verbatim quote still contains the fact)
      # partial: ≥ 60 % of the key value's word tokens (len ≥ 4) appear in the run value's text; numbers within 10 %
      # no: otherwise
  def match_claims(keys: Sequence[KeyClaim], run_claims: Sequence[Claim]) -> RecallReport
      # for each key: candidates = run claims whose field_key ∈ {key.field_key, *alternatives}; best verdict over candidates;
      # "field_only" when the field was written but no candidate value agrees; "missing" when no candidate exists
  def probook_keys(path: Path) -> list[KeyClaim]; async def zafran_keys(conn, *, workspace_id) -> list[KeyClaim]
  ```
