  function railColor(x){

      else if(x>centres[k]-band){i=k;t=smooth((x-(centres[k]-band))/(2*band));break;}

    var radios = document.querySelectorAll('input[type="radio"][name="mode-toggle"]');
    console.log('[mode-toggle] radios found:', radios.length);
    if (!radios.length) return;

    function updateActiveState() {
      radios.forEach(function(radio) {
        var wrapper = radio.closest('.mode-toggle-option');
        if (!wrapper) return;

        if (radio.checked) {
          wrapper.classList.add('is-active-mode');
        } else {
          wrapper.classList.remove('is-active-mode');
        }
      });

    // Ensure A is default if nothing is checked
    var checkedRadio = Array.from(radios).find(function(r) { return r.checked; });
    if (!checkedRadio) {
      checkedRadio = Array.from(radios).find(function(r) { return r.value === 'a'; });
      if (checkedRadio) {
        checkedRadio.checked = true;
      }
    }

    var defaultValue = checkedRadio ? checkedRadio.value : 'a';
    console.log('[mode-toggle] initial defaultValue:', defaultValue);

    if (defaultValue === 'a') {
      document.body.classList.add('mode-a');
      document.body.classList.remove('mode-b');
    } else if (defaultValue === 'b') {
      document.body.classList.add('mode-b');
      document.body.classList.remove('mode-a');
    }

    updateActiveState();

    // Listen for changes
    radios.forEach(function(radio) {
      radio.addEventListener('change', function() {

        var value = radio.value; // "a" or "b"

        if (value === 'a') {
          document.body.classList.add('mode-a');
          document.body.classList.remove('mode-b');
        } else if (value === 'b') {

      });
