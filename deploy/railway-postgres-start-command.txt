bash -c 'C=/var/lib/postgresql/data/certs; mkdir -p "$C"; [ -f "$C/server.key" ] || openssl req -new -x509 -days 3650 -nodes -subj "/CN=postgres" -keyout "$C/server.key" -out "$C/server.crt" 2>/dev/null; printf "local all all trust\nhost all all 127.0.0.1/32 trust\nhost all all ::1/128 trust\nhostssl all all all scram-sha-256\n" > "$C/pg_hba.conf"; chown -R postgres:postgres "$C"; chmod 600 "$C/server.key"; exec docker-entrypoint.sh postgres -c ssl=on -c ssl_cert_file=$C/server.crt -c ssl_key_file=$C/server.key -c hba_file=$C/pg_hba.conf -c max_connections=200 -c shared_buffers=1GB -c effective_cache_size=3GB -c work_mem=8MB -c maintenance_work_mem=256MB -c random_page_cost=1.1 -c idle_in_transaction_session_timeout=10min -c log_min_duration_statement=2s'
